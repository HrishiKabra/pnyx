% Preamble additions for the pnyx arXiv-style PDF build (paper/build.sh).
% Loaded via pandoc --include-in-header on top of its default LaTeX template.

% Single author line with no affiliation, per the writeup's byline.
\usepackage{authblk}
\renewcommand\Authfont{\normalsize}
\renewcommand\Affilfont{\itshape\small}

% We embed figures via raw \includegraphics in `{=latex}` fenced blocks
% (see paper/prepare.py), bypassing pandoc's own Image handling, so its
% automatic `\usepackage{graphicx}` guard never fires -- load it explicitly.
\usepackage{graphicx}

% The writeup is plain prose, not LaTeX: it uses literal Unicode symbols
% (rho, plus-minus, the minus sign, section markers, arrows, ...) rather than
% math-mode escapes. Map each one to the equivalent math-mode glyph so the
% PDF renders cleanly under xelatex regardless of the active text font's
% glyph coverage.
\usepackage{newunicodechar}
\newunicodechar{ρ}{\ensuremath{\rho}}
\newunicodechar{λ}{\ensuremath{\lambda}}
\newunicodechar{Δ}{\ensuremath{\Delta}}
\newunicodechar{Σ}{\ensuremath{\Sigma}}
\newunicodechar{Π}{\ensuremath{\Pi}}
\newunicodechar{±}{\ensuremath{\pm}}
\newunicodechar{×}{\ensuremath{\times}}
\newunicodechar{−}{\ensuremath{-}}
\newunicodechar{→}{\ensuremath{\rightarrow}}
\newunicodechar{⇒}{\ensuremath{\Rightarrow}}
\newunicodechar{∈}{\ensuremath{\in}}
\newunicodechar{·}{\ensuremath{\cdot}}
\newunicodechar{≈}{\ensuremath{\approx}}
\newunicodechar{≥}{\ensuremath{\geq}}
\newunicodechar{§}{\S}
